# ============================================================================
# DevFest 2026 - AI Microkernel : Full Runbook (clean slate -> running)
# ============================================================================
# USAGE: this is a runbook, not a script - do NOT `bash cmds.txt`.
#        Copy-paste one step at a time. Step 1b destroys the cluster and
#        Step 8 is a standalone alternative to Steps 5-7.
#
# Cluster facts for this machine (verified 2026-11-04):
#   minikube v1.39.0 | driver=docker | container-runtime=containerd | k8s v1.37.0
#   host: 8 CPU / 15.7 GB RAM
#
# NOTE: `kubectl` is NOT installed standalone on this machine.
#       Every command below uses `minikube kubectl --`.
#       Optional convenience:  alias kubectl='minikube kubectl --'
#
# IMPORTANT: this cluster runs the *containerd* runtime, NOT dockerd.
#       `eval $(minikube docker-env)` does NOT work here - minikube itself
#       warns it is "highly experimental" with containerd, and images built
#       into that daemon are invisible to the kubelet (-> ImagePullBackOff).
#       Use `minikube image load` (Step 4) instead.
# ============================================================================


# ----------------------------------------------------------------------------
# STEP 0 : Free port 8080 on the host
# ----------------------------------------------------------------------------
# A local `npm start` holds :8080 and will collide with both `docker run -p 8080`
# and the port-forward in Step 7. Find and stop it first.

ss -ltnp | grep ':8080'                 # identify the holder (look for `node`)
pkill -f 'node dist/index.js'           # stop local dev server + its MCP child
ss -ltnp | grep ':8080' || echo "8080 is free"


# ----------------------------------------------------------------------------
# STEP 1 : Nuke everything - full clean slate
# ----------------------------------------------------------------------------

# 1a. (Optional) soft teardown - just remove the app, keep the cluster:
minikube kubectl -- delete -f k8s/hpa.yaml --ignore-not-found
minikube kubectl -- delete -f k8s/service.yaml --ignore-not-found
minikube kubectl -- delete -f k8s/deployment.yaml --ignore-not-found

# 1b. HARD reset - destroys the cluster, all profiles, and ~/.minikube:
minikube delete --all --purge

# 1c. Verify nothing is left behind:
minikube profile list                   # expect: no valid profiles
docker ps -a | grep minikube            # expect: no rows

# 1d. (Optional) reclaim host disk from old image layers:
docker image rm devfest/ai-microkernel:1.0.0 2>/dev/null
docker system prune -f


# ----------------------------------------------------------------------------
# STEP 2 : Start a fresh cluster
# ----------------------------------------------------------------------------

minikube start \
  --driver=docker \
  --container-runtime=containerd \
  --kubernetes-version=v1.37.0 \
  --cpus=4 \
  --memory=4096

minikube status                         # host/kubelet/apiserver => Running
minikube kubectl -- get nodes           # node => Ready


# ----------------------------------------------------------------------------
# STEP 3 : Enable metrics-server (REQUIRED for the HPA)
# ----------------------------------------------------------------------------
# Without this the HPA reports `cpu: <unknown>/70%` and never scales.

minikube addons enable metrics-server
minikube kubectl -- wait --for=condition=ready pod \
  -l k8s-app=metrics-server -n kube-system --timeout=180s


# ----------------------------------------------------------------------------
# STEP 4 : Build the app and load it into the cluster
# ----------------------------------------------------------------------------

npm install
npm run build                           # tsc -> dist/

# Build on the HOST docker daemon:
docker build -t devfest/ai-microkernel:1.0.0 .

# Then push it into minikube's containerd store (this is the step that matters):
minikube image load devfest/ai-microkernel:1.0.0

# Verify the kubelet can actually see it:
minikube image ls | grep microkernel
# Alternative one-shot build that skips the load entirely:
#   minikube image build -t devfest/ai-microkernel:1.0.0 .


# ----------------------------------------------------------------------------
# STEP 5 : Deploy
# ----------------------------------------------------------------------------

minikube kubectl -- apply -f k8s/deployment.yaml
minikube kubectl -- apply -f k8s/service.yaml
minikube kubectl -- apply -f k8s/hpa.yaml

minikube kubectl -- rollout status deployment/ai-microkernel-deployment --timeout=180s


# ----------------------------------------------------------------------------
# STEP 6 : Verify
# ----------------------------------------------------------------------------

minikube kubectl -- get pods -l app=ai-microkernel       # expect 3x 1/1 Running
minikube kubectl -- get svc ai-microkernel-service
minikube kubectl -- get hpa                              # expect cpu: 0%/70%
minikube kubectl -- top pods -l app=ai-microkernel

# App logs (should show MCP server + listener lines):
minikube kubectl -- logs -l app=ai-microkernel --tail=20


# ----------------------------------------------------------------------------
# STEP 7 : Access and smoke-test
# ----------------------------------------------------------------------------

minikube kubectl -- port-forward svc/ai-microkernel-service 8080:80 &
# (use 18080:80 instead if anything still holds host :8080)

curl -s http://localhost:8080/healthz
# -> {"status":"UP"}

curl -s -X POST http://localhost:8080/api/v1/forecast \
  -H 'Content-Type: application/json' \
  -d '{"ambientNoiseDecibels":95,"lightingLux":1200,"timeSinceLastMealMinutes":300,"sugarIntakeGrams":60}'
# -> riskScore 100 / riskLevel "HIGH"

curl -s -X POST http://localhost:8080/api/v1/forecast \
  -H 'Content-Type: application/json' \
  -d '{"ambientNoiseDecibels":70,"lightingLux":500,"timeSinceLastMealMinutes":120,"sugarIntakeGrams":20}'
# -> riskScore 20 / riskLevel "LOW"

# Validation-failure path (bad input -> 400 + FAILED state):
curl -s -X POST http://localhost:8080/api/v1/forecast \
  -H 'Content-Type: application/json' -d '{"ambientNoiseDecibels":"loud"}'

# Stop the forward when done:
pkill -f 'port-forward svc/ai-microkernel-service'


# ----------------------------------------------------------------------------
# STEP 8 : Run the image standalone (no Kubernetes)
# ----------------------------------------------------------------------------

docker run --rm -p 18080:8080 --name mk-test devfest/ai-microkernel:1.0.0
# NOTE: left side is the HOST port. Use 18080 to dodge the :8080 collision.
# In another shell:  curl -s http://localhost:18080/healthz
docker stop mk-test


# ----------------------------------------------------------------------------
# STEP 9 : Redeploy after a code change
# ----------------------------------------------------------------------------

npm run build
docker build -t devfest/ai-microkernel:1.0.0 .
minikube image load devfest/ai-microkernel:1.0.0
minikube kubectl -- rollout restart deployment/ai-microkernel-deployment
minikube kubectl -- rollout status deployment/ai-microkernel-deployment --timeout=180s
# `rollout restart` is required: the tag is unchanged, so imagePullPolicy
# IfNotPresent means the running pods would otherwise keep the old layers.


# ----------------------------------------------------------------------------
# TROUBLESHOOTING - the three failures already hit on this project
# ----------------------------------------------------------------------------
# ImagePullBackOff
#   Image is not in minikube's containerd store. Re-run Step 4's
#   `minikube image load`. Do NOT rely on `eval $(minikube docker-env)`.
#
# CreateContainerConfigError
#   "container has runAsNonRoot and image has non-numeric user (node)"
#   The Dockerfile sets `USER node` (a name); the kubelet cannot verify a
#   name against runAsNonRoot. Fixed by runAsUser/runAsGroup 1000 in
#   k8s/deployment.yaml. Keep them in sync if the Dockerfile's USER changes.
#
# docker: "bind: address already in use"
#   Host :8080 is taken (usually a local `npm start`). See Step 0.
#
# HPA shows cpu: <unknown>/70%
#   metrics-server is disabled or still warming up. See Step 3; the first
#   scrape takes ~45s after the pod goes Ready.

# Useful digging commands:
minikube kubectl -- describe pod -l app=ai-microkernel
minikube kubectl -- get events --sort-by=.lastTimestamp | tail -20
minikube ssh -- "sudo crictl images | grep microkernel"
